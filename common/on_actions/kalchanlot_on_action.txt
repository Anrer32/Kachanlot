on_actions = {
    # Срабатывает на старте игры для всех стран
    on_startup = {
        effect = {
            every_country = {
                update_oil_factories = yes
            }
        }
    }
    
    # Обновляется раз в неделю для каждой страны (оптимально для производительности)
    on_monthly = {
        effect = {
            update_oil_factories = yes
        }
    }
}
